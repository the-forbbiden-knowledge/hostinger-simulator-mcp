cmake_minimum_required(VERSION 3.24)
project(hostinger_simulator_native LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

add_library(simulator_core STATIC src/native/simulator.cpp src/native/agent_security.cpp src/native/recon_security.cpp src/native/capstone_security.cpp src/native/supplychain_security.cpp src/native/vector_security.cpp src/native/a2a_security.cpp src/native/threat_model.cpp src/native/infrastructure_security.cpp src/native/mcp_tool_security.cpp src/native/rag_security.cpp src/native/ai_assurance.cpp src/native/service_policy.cpp src/native/change_security.cpp src/native/assessment_lifecycle.cpp)
target_include_directories(simulator_core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)

if(MSVC)
  target_compile_options(simulator_core PRIVATE /W4 /permissive- /sdl /EHsc)
else()
  target_compile_options(simulator_core PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
endif()

add_executable(simulator src/native/main.cpp)
target_link_libraries(simulator PRIVATE simulator_core)
